```python
#!/usr/bin/env python3
import operator

ops = {
    '+': operator.add,
    '-': operator.sub,
    '*': operator.mul,
    '/': operator.truediv,
    '^': operator.pow,
}

def calc(expr: str):
    for sym in ops:
        if sym in expr:
            left, right = expr.split(sym, 1)
            try:
                a, b = float(left.strip()), float(right.strip())
                return ops[sym](a, b)
            except ValueError:
                return "Invalid numbers"
    return "Unsupported operation"

def repl():
    print("Simple Calculator – type 'exit' to quit")
    while True:
        expr = input("> ")
        if expr.lower() in ('exit', 'quit'):
            break
        print(calc(expr))

if __name__ == "__main__":
    repl()
```

[2026-09-14 09:35:22] UK: Ek python code do jo  browese se camera info lele face track kare real time
[2026-09-14 09:35:22] JARVIS: Namaste sir, yeh ek simple Python script hai jo OpenCV ke `VideoCapture` se webcam ka feed leke MediaPipe ka face‑mesh use karke real‑time face tracking karta hai. Aap isse VS Code ya MT Manager mein run kar sakte hain.

```python
# pip install opencv-python mediapipe
import cv2
import mediapipe as mp

# MediaPipe face mesh setup
mp_face_mesh = mp.solutions.face_mesh
face_mesh = mp_face_mesh.FaceMesh(
    static_image_mode=False,
    max_num_faces=1,
    refine_landmarks=True,
    min_detection_confidence=0.5,
    min_tracking_confidence=0.5,
)

mp_drawing = mp.solutions.drawing_utils
drawing_spec = mp_drawing.DrawingSpec(thickness=1, circle_radius=1, color=(0,255,0))

# Webcam se video capture
cap = cv2.VideoCapture(0)   # 0 = default camera, change if needed

while cap.isOpened():
    ret, frame = cap.read()
    if not ret:
        break

    # BGR se RGB conversion
    rgb_frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
    results = face_mesh.process(rgb_frame)

    # Agar face detect hua toh landmarks draw karo
    if results.multi_face_landmarks:
        for face_landmarks in results.multi_face_landmarks:
            mp_drawing.draw_landmarks(
                image=frame,
                landmark_list=face_landmarks,
                connections=mp_face_mesh.FACEMESH_TESSELATION,
                landmark_drawing_spec=drawing_spec,
                connection_drawing_spec=drawing_spec,
            )

    cv2.imshow('Real‑time Face Tracking', frame)

    # 'q' dabane par exit
    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

4. Webcam ka feed window mein aayega; face ke landmarks real‑time update honge.

def sub(a, b):
    return a - b

def mul(a, b):
    return a * b

def div(a, b):
    if b == 0:
        return "Error: Division by zero"
    return a / b

def calculator():
    print("Simple Calculator")
    print("Operations: +  -  *  /")
    while True:
        expr = input("Enter expression (or 'exit' to quit): ")
        if expr.lower() == 'exit':
            break
        try:
            a, op, b = expr.split()
            a, b = float(a), float(b)
            if op == '+':
                result = add(a, b)
            elif op == '-':
                result = sub(a, b)
            elif op == '*':
                result = mul(a, b)
            elif op == '/':
                result = div(a, b)
            else:
                result = "Invalid operator"
            print("Result:", result)
        except Exception as e:
            print("Error:", e)

if __name__ == "__main__":
